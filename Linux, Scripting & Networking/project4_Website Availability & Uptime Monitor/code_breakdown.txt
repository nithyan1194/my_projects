#!/bin/bash
# ==============================================================================
# Script Name:    uptime_monitor.sh
# Description:    Automated website availability and response time monitor.
# Tier:           Intermediate DevOps Scripting
# ==============================================================================

# Configuration
readonly TARGETS_FILE="${1:-targets.txt}"
readonly LOG_FILE="uptime_audit.log"
readonly TIMEOUT=5 # seconds

*********************************************************
1. Header & Configuration Setup
#!/bin/bash: The shebang pointing to the Bash shell interpreter.
readonly TARGETS_FILE="${1:-targets.txt}": Locks the target variable. It takes the first CLI parameter ($1) as the target list file, but defaults to targets.txt if no argument is passed.
readonly LOG_FILE="uptime_audit.log": Defines an append-only file path to store execution telemetry.
readonly TIMEOUT=5: Sets a strict 5-second HTTP request limit to prevent hanging threads.
*********************************************************
2. Pre-flight Guard Clause
# Input Validation
if [ ! -f "$TARGETS_FILE" ]; then
    echo "Error: Targets file '$TARGETS_FILE' not found." >&2
    echo "Usage: $0 [path_to_targets_file]" >&2
    exit 1
fi
*********************************************************
[ ! -f "$TARGETS_FILE" ]: Checks if the target file does not exist or is invalid.
>&2: Redirects error outputs specifically to Standard Error (stderr) so error text isn't mixed into standard data pipelines.
exit 1: Halts execution immediately with an error exit code.
*********************************************************
TIMESTAMP=$(date +"%Y-%m-%d %H:%M:%S")

echo "=========================================================="
echo "          WEBSITE AVAILABILITY CHECK REPORT"
echo "=========================================================="
echo "Execution Time: $TIMESTAMP"
echo "Target List:    $TARGETS_FILE"
echo "----------------------------------------------------------"

while IFS= read -r url || [ -n "$url" ]; do
    # Skip empty lines and comments
    [[ -z "$url" || "$url" =~ ^# ]] && continue
*********************************************************
3. File Stream Ingestion Loop
while IFS= read -r url: Streams the file line by line:
IFS=: Preserves leading/trailing whitespaces.
-r: Prevents backslash escapes from altering the URL string.
|| [ -n "$url" ]: Ensures the last line is processed even if it doesn't end with a newline character.
[[ -z "$url" || "$url" =~ ^# ]]: Defensive check:
-z "$url": Ignores empty blank lines.
"$url" =~ ^#: Ignores comment lines starting with #.
continue: Skips to the next line immediately if either condition is met.
*********************************************************
    # Measure HTTP status code and total response time (in seconds)
    response=$(curl -s -o /dev/null -w "%{http_code} %{time_total}" --max-time "$TIMEOUT" "$url")

    status_code=$(echo "$response" | awk '{print $1}')
    response_time=$(echo "$response" | awk '{print $2}')
*********************************************************
4. High-Speed Probe & Telemetry Extraction

curl -s: Silent mode—hides curl's built-in progress bars and error output.
-o /dev/null: Redirects the actual HTML/JSON web page content to the null device, throwing it away so we don't pollute local disk space.
-w "%{http_code} %{time_total}": Write out metric variables. Requests curl to return only the HTTP status code (e.g., 200, 404) and total latency time in seconds (e.g., 0.142).
--max-time "$TIMEOUT": Aborts the HTTP connection attempt if the target host takes longer than 5 seconds to respond.

5. String Extraction via awk

awk '{print $1}': Isolates column 1 from the curl output string, assigning the HTTP response code to status_code.
awk '{print $2}': Isolates column 2, assigning the total network response latency (in seconds) to response_time.
*********************************************************
    # Validate response status
    if [ "$status_code" -eq 200 ]; then
        status="UP"
        echo -e "[SUCCESS] $url | Status: $status_code | Latency: ${response_time}s"
    elif [ "$status_code" -eq 000 ]; then
        status="TIMEOUT/DOWN"
        echo -e "[CRITICAL] $url | Connection Timed Out (> ${TIMEOUT}s)" >&2
    else
        status="ERROR"
        echo -e "[WARNING]  $url | Status: $status_code | Latency: ${response_time}s" >&2
    fi
*********************************************************
6. Health State Evaluation
$status_code -eq 200: Marks successful web server responses.
$status_code -eq 000: A status of 000 is generated by curl when a network timeout occurs, a DNS resolution fails, or a server is completely unreachable.
else: Flags non-200 responses (such as 404 Not Found or 500 Server Error).

7. Audit Logging & Loop Termination

>> "$LOG_FILE": Appends timestamped operational telemetry to the audit ledger.
done < "$TARGETS_FILE": Feeds the input file directly into the while loop stream.

*********************************************************
    # Append telemetry data to audit log
    echo "[$TIMESTAMP] URL: $url | Status: $status ($status_code) | Response Time: ${response_time}s" >> "$LOG_FILE"

done < "$TARGETS_FILE"

echo "=========================================================="
echo "Telemetry recorded to: $LOG_FILE"
